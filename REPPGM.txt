       IDENTIFICATION DIVISION.
       PROGRAM-ID. REPPGM.
       ENVIRONMENT DIVISION.
       INPUT-OUTPUT SECTION.
       FILE-CONTROL.
           SELECT TI001-CUST-IN ASSIGN TO CUSTIN
             ACCESS       IS SEQUENTIAL
             ORGANIZATION IS SEQUENTIAL
             FILE STATUS  IS WS05-FST-IN.
           SELECT TO001-REP ASSIGN TO REP
             ACCESS       IS SEQUENTIAL
             ORGANIZATION IS SEQUENTIAL
             FILE STATUS  IS WS05-FST-OUT.
       DATA DIVISION.
       FILE SECTION.
       FD  TI001-CUST-IN.
      *    RECORD CONTAINS 160 CHARACTERS.
       01  INREC.
           05 TI001-HID     PIC X(5).
           05 FILLER        PIC X.
           05 TI001-NAME    PIC X(10).
           05 FILLER        PIC X.
           05 TI001-STREET  PIC X(20).
           05 FILLER        PIC X.
           05 TI001-CITY    PIC X(20).
           05 FILLER        PIC X.
           05 TI001-STATE   PIC X(20).
           05 FILLER        PIC X.
           05 TI001-PIN     PIC 9(6).
           05 FILLER        PIC X.
           05 TI001-PHONE   PIC 9(10).
           05 FILLER        PIC X.
           05 TI001-PLY-NO  PIC 9(10).
           05 FILLER        PIC X.
           05 TI001-PLY-AMT PIC 9(10).
           05 FILLER        PIC X.
           05 TI001-DOB     PIC X(10).
           05 FILLER        PIC X.
           05 TI001-PLY-ID  PIC X(5).
           05 FILLER        PIC X.
           05 TI001-PLY-CODE PIC X(10).
           05 FILLER        PIC X.
           05 TI001-AGT-CODE PIC X(5).
           05 FILLER        PIC X(7).
       FD  TO001-REP.
       01  TO001-REP-REC    PIC X(80).
       WORKING-STORAGE SECTION.
       01  WS01-HEADER-1.
           05 FILLER                PIC X(19).
           05 WS05-TITLE            PIC X(30).
           05 FILLER                PIC X(31).
       01  WS01-HEADER-1A.
           05 USERID                PIC X(8)  VALUE 'USER-ID:'.
           05 WS05-USERID           PIC X(9).
           05 FILLER                PIC X(33).
           05 USER-NAME             PIC X(7)  VALUE 'NAME : '.
           05 WS05-USER-NAME        PIC X(10) VALUE 'SUPRAJA'.
           05 FILLER                PIC X(13).
       01  WS01-HEADER-2A.
           05 DATEE                 PIC X(7)  VALUE 'DATE : '.
           05 WS05-DATE             PIC X(10).
           05 FILLER                PIC X(33).
           05 PAGE-NO               PIC X(7)  VALUE 'PAGE : '.
           05 WS05-PAGE-NO          PIC 9(03) VALUE 0.
           05 FILLER                PIC X(20).
       01  WS01-HEADER-3A.
           05 TIMEE                 PIC X(7)  VALUE 'TIME : '.
           05 WS05-TIME             PIC X(10).
           05 FILLER                PIC X(33).
           05 REPORT-ID             PIC X(16) VALUE 'REPORT ID : 1012'.
           05 FILLER                PIC X(14).
       01  WS01-HEADER-2B.
           05 FILLER                PIC X(55) VALUE SPACES.
           05 PAGE-NO-1             PIC X(5)  VALUE 'PAGE:'.
           05 WS05-PAGE-NO-1        PIC 9(03).
           05 FILLER                PIC X(17).
       01  WS01-HEADER-3B.
           05 FILLER                PIC X(55) VALUE SPACES.
           05 REPORT-ID1            PIC X(10) VALUE 'REPORT ID:'.
           05 FILLER                PIC X(4)  VALUE '1012'.
           05 FILLER                PIC X(11).
       01  WS01-HEADER-4.
           05 FILLER                PIC X(80) VALUE SPACES.
       01  WS01-HEADER-5.
           05 TITLE-1               PIC X(09) VALUE 'HOLDER ID'.
           05 FILLER                PIC X.
           05 TITLE-2               PIC X(11) VALUE 'HOLDER NAME'.
           05 FILLER                PIC X.
           05 TITLE-3               PIC X(10) VALUE 'DOB'.
           05 FILLER                PIC X.
           05 TITLE-4               PIC X(10) VALUE 'AGENT CODE'.
           05 FILLER                PIC X(37).
       01  WS01-HEADER-6.
           05 TITLE-1               PIC X(09) VALUE ALL '-'.
           05 FILLER                PIC X.
           05 TITLE-2               PIC X(11) VALUE ALL '-'.
           05 FILLER                PIC X.
           05 TITLE-3               PIC X(10) VALUE ALL '-'.
           05 FILLER                PIC X.
           05 TITLE-4               PIC X(10) VALUE ALL '-'.
           05 FILLER                PIC X(37).
           05 FILLER                PIC X(80) VALUE ALL '-'.
       01  WS01-RECORDS.
           05 WS05-HOLDER-ID        PIC X(9).
           05 FILLER                PIC X.
           05 WS05-HOLDER-NAME      PIC X(11).
           05 FILLER                PIC X.
           05 WS05-DOB              PIC X(10).
           05 FILLER                PIC X.
           05 WS05-AGT-CODE         PIC X(10).
           05 FILLER                PIC X(37).
       01  WS01-FOOTER-1.
           05 SYMBOL-6              PIC X(31) VALUE ALL '-'.
           05 END-OF-PAGES          PIC X(11) VALUE 'END OF PAGE'.
           05 SYMBOL-7              PIC X(30) VALUE ALL '-'.
           05 FILLER                PIC X(08).
       01  WS01-FOOTER-2.
           05 SYMBOL-8              PIC X(30) VALUE ALL '*'.
           05 END-OF-REPORT         PIC X(13) VALUE 'END OF REPORT'.
           05 SYMBOL-9              PIC X(29) VALUE ALL '*'.
           05 FILLER                PIC X(08).
       01  WS01-VARS.
           05 WS05-FST-IN           PIC 9(02) VALUE 0.
           05 WS05-FST-OUT          PIC 9(02) VALUE 0.
           05 WS05-REC-COUNT        PIC 9(05) VALUE 0.
           05 WS05-LINE-NO          PIC 9(02) VALUE 1.
           05 WS05-DATE-TIME.
               10 WS10-YEAR         PIC 9(04).
               10 WS10-MONTH        PIC 9(02).
               10 WS10-DAY          PIC 9(02).
               10 WS10-HOUR         PIC 9(02).
               10 WS10-MINS         PIC 9(02).
               10 WS10-SECS         PIC 9(02).
       LINKAGE SECTION.
       01 LK-PARM PIC X(10).
       PROCEDURE DIVISION USING LK-PARM.
       0000-MAIN-PARA.
            PERFORM 1000-INIT-PARA.
            PERFORM 3000-PROC-PARA THRU
                    3000-PROC-PARA-EXIT.
            PERFORM 9000-TERM-PARA.
       1000-INIT-PARA.
            CONTINUE.
       3000-PROC-PARA.
            PERFORM 3100-OPEN-PARA THRU
                    3100-OPEN-PARA-EXIT.
            PERFORM 3200-HEADER-PARA THRU
                    3200-HEADER-PARA-EXIT.
            PERFORM 3300-READ-PARA THRU
                    3300-READ-PARA-EXIT UNTIL WS05-FST-IN = 10.
            PERFORM 3400-FOOTER-PARA THRU
                    3400-FOOTER-PARA-EXIT.
            PERFORM 3500-CLOSE-PARA THRU
                    3500-CLOSE-PARA-EXIT.
       3000-PROC-PARA-EXIT.
            EXIT.
       3100-OPEN-PARA.
            OPEN INPUT TI001-CUST-IN
            EVALUATE TRUE
             WHEN WS05-FST-IN = 00
                 DISPLAY 'TI001-CUST-IN WAS OPENED'
             WHEN OTHER
                 DISPLAY 'TI001-CUST-IN WAS NOT OPENED'
                 DISPLAY 'FILE STATUS : ' WS05-FST-IN
                 PERFORM 9000-TERM-PARA
            END-EVALUATE
            OPEN OUTPUT TO001-REP
            EVALUATE TRUE
             WHEN WS05-FST-OUT = 00
                 DISPLAY 'TO001-REP WAS OPENED'
             WHEN OTHER
                 DISPLAY 'TO001-REP WAS NOT OPENED'
                 DISPLAY 'FILE STATUS : ' WS05-FST-OUT
                 PERFORM 9000-TERM-PARA
            END-EVALUATE.
       3100-OPEN-PARA-EXIT.
            EXIT.
       3200-HEADER-PARA.
            MOVE LK-PARM TO WS05-USERID
            MOVE 'CHILD INSURANCE DETAIL REPORT' TO WS05-TITLE
            MOVE FUNCTION CURRENT-DATE TO WS05-DATE-TIME
            MOVE WS10-YEAR  TO WS05-DATE(1:4)
            MOVE '/'        TO WS05-DATE(5:1)
            MOVE WS10-MONTH TO WS05-DATE(6:2)
            MOVE '/'        TO WS05-DATE(8:1)
            MOVE WS10-DAY   TO WS05-DATE(9:2)
            MOVE WS10-HOUR  TO WS05-TIME(1:2)
            MOVE ':'        TO WS05-TIME(3:1)
            MOVE WS10-MINS  TO WS05-TIME(4:2)
            MOVE ':'        TO WS05-TIME(6:1)
            MOVE WS10-SECS  TO WS05-TIME(7:2)
            ADD 1 TO WS05-PAGE-NO
            WRITE TO001-REP-REC FROM WS01-HEADER-1
            WRITE TO001-REP-REC FROM WS01-HEADER-1A
            WRITE TO001-REP-REC FROM WS01-HEADER-2A
            WRITE TO001-REP-REC FROM WS01-HEADER-3A
            WRITE TO001-REP-REC FROM WS01-HEADER-4
            WRITE TO001-REP-REC FROM WS01-HEADER-5
            WRITE TO001-REP-REC FROM WS01-HEADER-6
            EVALUATE TRUE
             WHEN WS05-FST-OUT = 00
                 DISPLAY 'HEADER WAS WRITTEN'
             WHEN OTHER
                 DISPLAY 'HEADER WAS NOT WRITTEN'
                 DISPLAY 'FILE STATUS : ' WS05-FST-OUT
                 PERFORM 9000-TERM-PARA
            END-EVALUATE.
       3200-HEADER-PARA-EXIT.
            EXIT.
       3300-READ-PARA.
            MOVE SPACES TO INREC
            ADD 1 TO WS05-REC-COUNT
            READ TI001-CUST-IN
            EVALUATE TRUE
             WHEN WS05-FST-IN = 00
                 DISPLAY 'RECORD ' WS05-REC-COUNT ' WAS READ'
                 PERFORM 3310-WRITE-PARA THRU
                         3310-WRITE-PARA-EXIT
             WHEN WS05-FST-IN = 10
                 DISPLAY 'ALL RECORDS WERE READ'
             WHEN OTHER
                 DISPLAY 'RECORD ' WS05-REC-COUNT ' WAS UNABLE TO READ'
                 DISPLAY 'FILE STATUS : ' WS05-FST-IN
            END-EVALUATE.
       3300-READ-PARA-EXIT.
            EXIT.
       3310-WRITE-PARA.
            MOVE TI001-HID   TO WS05-HOLDER-ID
            MOVE TI001-NAME  TO WS05-HOLDER-NAME
            MOVE TI001-DOB   TO WS05-DOB
            MOVE TI001-AGT-CODE TO WS05-AGT-CODE
            EVALUATE TRUE
             WHEN WS05-LINE-NO >= 1 AND WS05-LINE-NO < 13
                 WRITE TO001-REP-REC FROM WS01-RECORDS
              EVALUATE TRUE
               WHEN WS05-FST-OUT = 00
                   DISPLAY 'RECORD ' WS05-REC-COUNT ' WAS WRITTEN'
               WHEN OTHER
                   DISPLAY 'RECORD ' WS05-REC-COUNT ' WAS NOT WRITTEN'
                   DISPLAY 'FILE STATUS : ' WS05-FST-OUT
              END-EVALUATE
               ADD 1 TO WS05-LINE-NO
             WHEN WS05-LINE-NO = 13
               WRITE TO001-REP-REC FROM WS01-RECORDS
               WRITE TO001-REP-REC FROM WS01-FOOTER-1
                 ADD 1 TO WS05-PAGE-NO
                MOVE WS05-PAGE-NO TO WS05-PAGE-NO-1
               WRITE TO001-REP-REC FROM WS01-HEADER-2B
               WRITE TO001-REP-REC FROM WS01-HEADER-3B
               WRITE TO001-REP-REC FROM WS01-HEADER-4
               WRITE TO001-REP-REC FROM WS01-HEADER-5
               WRITE TO001-REP-REC FROM WS01-HEADER-6
               MOVE 1 TO WS05-LINE-NO
               EVALUATE TRUE
                WHEN WS05-FST-OUT = 00
                   DISPLAY 'RECORD ' WS05-REC-COUNT ' WAS WRITTEN'
                WHEN OTHER
                   DISPLAY 'RECORD ' WS05-REC-COUNT ' WAS NOT WRITTEN'
                   DISPLAY 'FILE STATUS : ' WS05-FST-OUT
               END-EVALUATE
               DISPLAY 'END OF PAGE : ' WS05-PAGE-NO
            END-EVALUATE.
       3310-WRITE-PARA-EXIT.
            EXIT.
       3400-FOOTER-PARA.
            WRITE TO001-REP-REC FROM WS01-FOOTER-2
            EVALUATE TRUE
             WHEN WS05-FST-OUT = 00
                 DISPLAY 'FOOTER WAS WRITTEN'
             WHEN OTHER
                 DISPLAY 'FOOTER WAS NOT WRITTEN'
                 DISPLAY 'FILE STATUS : ' WS05-FST-OUT
            END-EVALUATE.
       3400-FOOTER-PARA-EXIT.
            EXIT.
       3500-CLOSE-PARA.
            CLOSE TI001-CUST-IN TO001-REP.
       3500-CLOSE-PARA-EXIT.
            EXIT.
       9000-TERM-PARA.
            STOP RUN.
