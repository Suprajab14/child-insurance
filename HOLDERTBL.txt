DROP TABLE HOLDERTBL;
COMMIT;
CREATE TABLE HOLDERTBL (
    HOLDER_ID CHAR(5) NOT NULL PRIMARY KEY,
    HOLDER_NAME CHAR(10),
    STREET CHAR(20),
    CITY CHAR(20),
    STATE CHAR(20),
    PIN_CODE CHAR(06),
    PHONE_NUMBER CHAR(12),
    POLICY_NUMBER CHAR(11),
    POLICY_AMOUNT DECIMAL(10,2),
    DATE_OF_BIRTH CHAR(10),
    POLICY_ID CHAR(5),
    POLICY_CODE CHAR(10),
    AGENT_CODE CHAR(5))
    IN SHRDB4.SHRTS4;
CREATE UNIQUE INDEX HLDIDX ON HOLDERTBL(HOLDER_ID);
COMMIT;
